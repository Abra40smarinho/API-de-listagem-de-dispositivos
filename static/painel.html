// ---------- Lógica Analítica de Saúde e Ocorrências (Integrada com SQLite Tuya) ---------- const SC = { urlDiaria: '/api/disponibilidade-diaria', refreshMs: 60000, }; const SC_SERIES = [ { key: 'energia', label: 'Possível Queda de Energia', cor: '#fb7185' }, { key: 'rede', label: 'Queda de Rede / Comunicação', cor: '#8da2f0' }, { key: 'reconexao', label: 'Reconexão', cor: '#4aa8f0' }, ]; let scChart = null, scHoras = [], scOcorrencias = []; function fmtDuracao(segundos) { const t = Number(segundos); if (t <= 0) return '1s'; if (t < 60) return `${Math.round(t)}s`; if (t < 3600) { const m = Math.floor(t / 60), s = Math.round(t % 60); return s ? `${m}m ${s}s` : `${m}m`; } const h = Math.floor(t / 3600), m = Math.floor((t % 3600) / 60); return m ? `${h}h ${m}m` : `${h}h`; } const fmtDataHora = tsStr => { if (!tsStr) return '—'; const d = new Date(tsStr); return isNaN(d.getTime()) ? tsStr : d.toLocaleString('pt-BR'); }; function scProcessarEventos(eventos) { const t1 = Date.now(); const ini = Math.floor(t1 / 1000 / 3600) * 3600 - 23 * 3600; // Cria as 24 barras das últimas horas const horas = Array.from({ length: 24 }, (_, i) => ({ ini: ini + i * 3600, energia: 0, rede: 0, reconexao: 0 })); const add = (tsSec, k) => { const i = Math.floor((tsSec - ini) / 3600); if (i >= 0 && i < 24) horas[i][k]++; }; let todasOcorrencias = []; let totalEnergia = 0; let totalRede = 0; eventos.forEach(ev => { const tsSec = Math.floor(new Date(ev.timestamp).getTime() / 1000); const duracao = ev.duracao_seg || 300; // Padrão 5 min do polling const isEnergia = (ev.tipo || '').toLowerCase().includes('energia') || (ev.tipo || '').includes('2 entidades'); if (isEnergia) { totalEnergia++; add(tsSec, 'energia'); } else { totalRede++; add(tsSec, 'rede'); } todasOcorrencias.push({ entity_id: ev.entity_id, inicio: tsSec, duracao: duracao, tipo: isEnergia ? 'Possível queda de energia' : 'Queda de rede / comunicação' }); }); scHoras = horas; scOcorrencias = todasOcorrencias.sort((a, b) => b.inicio - a.inicio); // Atualiza os contadores dos cards superiores $('stat-energia').textContent = totalEnergia; $('stat-rede').textContent = totalRede; $('stat-reconexao').textContent = scOcorrencias.length; } function scRenderGrafico() { if (!window.Chart) return; const labels = scHoras.map(h => new Date(h.ini * 1000).toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' })); const dados = SC_SERIES.map(s => scHoras.map(h => h[s.key] || 0)); if (!scChart) { scChart = new Chart($('sc-canvas'), { type: 'bar', data: { labels, datasets: SC_SERIES.map((s, i) => ({ label: s.label, data: dados[i], backgroundColor: s.cor, borderWidth: 0, stack: 'ev', categoryPercentage: 0.9, barPercentage: 0.9 })), }, options: { responsive: true, maintainAspectRatio: false, animation: false, interaction: { mode: 'index', intersect: false }, plugins: { legend: { position: 'top', labels: { color: '#94a3b8', boxWidth: 10, boxHeight: 10, padding: 10, font: { size: 11 } } }, tooltip: { backgroundColor: '#0f172a', borderColor: '#334155', borderWidth: 1, titleColor: '#e2e8f0', bodyColor: '#cbd5e1', callbacks: { title: it => { const h = scHoras[it[0].dataIndex]; return `${new Date(h.ini*1000).toLocaleTimeString('pt-BR', {hour:'2-digit', minute:'2-digit'})}`; } } } }, scales: { x: { stacked: true, grid: { display: false }, ticks: { color: '#94a3b8', font: { size: 10 }, maxRotation: 0 } }, y: { stacked: true, beginAtZero: true, ticks: { color: '#94a3b8', precision: 0, font: { size: 10 } }, grid: { color: 'rgba(203,213,225,.1)' } } } } }); } else { scChart.data.labels = labels; scChart.data.datasets.forEach((ds, i) => { ds.data = dados[i]; }); scChart.update(); } } function scRenderTabelaLogs() { const tbody = $('sc-tabela-logs'); if (!scOcorrencias.length) { tbody.innerHTML = `Nenhum evento de instabilidade registrado nas últimas 24h.`; return; } tbody.innerHTML = scOcorrencias.slice(0, 50).map(ev => { let corBadge = 'bg-indigo-500/10 text-indigo-400 border-indigo-500/20'; if (ev.tipo.includes('energia')) corBadge = 'bg-rose-500/10 text-rose-400 border-rose-500/20'; return ` ${fmtDataHora(ev.inicio * 1000)} ${esc(ev.entity_id)} ${esc(ev.tipo)} ${fmtDuracao(ev.duracao)} Indisponível por ${fmtDuracao(ev.duracao)} `; }).join(''); } async function carregarSaudeConexao() { try { $('sc-resumo').textContent = 'Carregando análise...'; const res = await fetch(SC.urlDiaria); if (!res.ok) throw new Error(`Erro HTTP: ${res.status}`); const j = await res.json(); const eventos = j.eventos || []; scProcessarEventos(eventos); scRenderGrafico(); scRenderTabelaLogs(); $('sc-resumo').textContent = `${scOcorrencias.length} eventos mapeados nas últimas 24h`; } catch (e) { console.error('Erro ao carregar diagnóstico de saúde:', e); $('sc-resumo').textContent = 'Erro ao carregar dados'; $('sc-tabela-logs').innerHTML = `Erro ao carregar o diagnóstico analítico da Tuya.`; } } $('btnAtualizarSaude').onclick = carregarSaudeConexao; carregarSaudeConexao(); setInterval(carregarSaudeConexao, SC.refreshMs);
